\subsection{ELBM Solver}

\begin{lstlisting}[language=C++]
void ELBMSolver::collide() {
    for (int y = 0; y < ny; ++y) {
        for (int x = 0; x < nx; ++x) {
            auto feq = computeEntropicEquilibrium(rho(x,y), u(x,y));
            auto [alpha_min, alpha_max] = computeAlphaBounds(f, feq);
            double alpha = solveAlpha(f, feq, alpha_min, alpha_max);
            double beta = 1.0 / (alpha * (3.0*nu + 0.5));

            // Two-step collision
            for (int i = 0; i < Q; ++i) {
                double fstar = f(x,y,i) + alpha*(feq[i] - f(x,y,i));
                f(x,y,i) = (1-beta)*f(x,y,i) + beta*fstar;
            }
        }
    }
}
\end{lstlisting}

\section{Alpha Solver (Newton-Raphson)}

\begin{lstlisting}[language=C++]
double solveAlpha(const auto& f, const auto& feq,
                  double alpha_min, double alpha_max) {
    double alpha = 2.0;  // Initial guess
    double H0 = computeH(f);

    for (int iter = 0; iter < 20; ++iter) {
        double R = 0.0, dR = 0.0;
        for (int i = 0; i < Q; ++i) {
            double f_alpha = f[i] + alpha*(feq[i] - f[i]);
            R += f_alpha * log(f_alpha / w[i]);
            dR += (feq[i] - f[i]) * (log(f_alpha/w[i]) + 1);
        }
        R -= H0;

        if (abs(R) < 1e-12) break;
        alpha = clamp(alpha - R/dR, alpha_min, alpha_max);
    }
    return alpha;
}
\end{lstlisting}

\section{Computational Workflow}

The simulation loop follows strict ordering:
\begin{enumerate}
    \item Compute macroscopic variables: $\rho, \mathbf{u}$ from $\mathbf{f}$
    \item Collision: Apply BGK or ELBM operator
    \item Streaming: $f_i(\mathbf{x}+\mathbf{c}_i) \leftarrow f_i(\mathbf{x})$
    \item Boundary conditions: Apply \textit{after} streaming
\end{enumerate}

\textbf{Critical}: Boundary conditions must be applied after streaming, not before, to avoid corruption.
